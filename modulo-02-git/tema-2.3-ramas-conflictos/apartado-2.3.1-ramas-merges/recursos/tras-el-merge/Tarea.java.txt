package curso.git;

/**
 * Tarea con identificador, título, descripción y estado.
 *
 * Toda tarea nueva empieza en estado PENDIENTE.
 */
public class Tarea {

    private final int id;
    private final String titulo;
    private final String descripcion;
    private EstadoTarea estado;

    public Tarea(int id, String titulo, String descripcion) {
        this.id = id;
        this.titulo = titulo;
        this.descripcion = descripcion;
        this.estado = EstadoTarea.PENDIENTE;
    }

    public int getId() {
        return id;
    }

    public String getTitulo() {
        return titulo;
    }

    public String getDescripcion() {
        return descripcion;
    }

    public EstadoTarea getEstado() {
        return estado;
    }

    public void setEstado(EstadoTarea estado) {
        this.estado = estado;
    }

    /**
     * Devuelve una línea de texto con los datos principales de la tarea.
     */
    public String resumen() {
        return String.format("#%-2d %-22s [%s] - %s", id, titulo, estado, descripcion);
    }

    @Override
    public String toString() {
        return resumen();
    }
}
